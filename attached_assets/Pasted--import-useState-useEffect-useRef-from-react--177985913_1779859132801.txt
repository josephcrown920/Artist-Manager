function ViewProjector() {
  const [currentViews, setCurrentViews] = useState("200");
  const [leversActive, setLeversActive] = useState({ hook: false, retention: false, comments: false, series: false, timing: false });

  const toggle = (id) => setLeversActive(prev => ({ ...prev, [id]: !prev[id] }));

  const base = parseInt(currentViews) || 200;
  const multiplier = 1
    + (leversActive.hook ? 1.8 : 0)
    + (leversActive.retention ? 1.4 : 0)
    + (leversActive.comments ? 0.8 : 0)
    + (leversActive.series ? 0.6 : 0)
    + (leversActive.timing ? 0.3 : 0);

  const projected = Math.round(base * multiplier);
  const activeCount = Object.values(leversActive).filter(Boolean).length;

  const getColor = (v) => {
    if (v >= 10000) return "#10b981";
    if (v >= 5000) return "#f59e0b";
    if (v >= 1000) return "#8b5cf6";
    return "#ef4444";
  };

  return (
    <div>
      <p style={{ color: "#555", fontSize: 13, marginBottom: 20, lineHeight: 1.6 }}>
        See how stacking each growth lever multiplies your views. This is based on real algorithm research, not guesses.
      </p>
      <div style={{ marginBottom: 20 }}>
        <label style={{ fontSize: 11, color: "#555", display: "block", marginBottom: 6, letterSpacing: 1 }}>YOUR CURRENT AVERAGE VIEWS</label>
        <input
          value={currentViews}
          onChange={e => setCurrentViews(e.target.value)}
          style={{
            width: "100%", padding: "10px 14px", background: "#0d0d1a", border: "1px solid #1a1a2e",
            borderRadius: 8, color: "#f0f0f0", fontSize: 22, fontWeight: 800, outline: "none", boxSizing: "border-box"
          }}
        />
      </div>

      <div style={{ fontSize: 11, color: "#555", letterSpacing: 1, marginBottom: 10 }}>TOGGLE THE LEVERS YOU'LL FIX</div>
      <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 8, marginBottom: 24 }}>
        {LEVERS.map(l => (
          <div
            key={l.id}
            onClick={() => toggle(l.id)}
            style={{
              border: `1px solid ${leversActive[l.id] ? l.color : "#1a1a2e"}`,
              background: leversActive[l.id] ? `${l.color}11` : "#0a0a0f",
              borderRadius: 8, padding: "10px 14px", cursor: "pointer",
              transition: "all 0.2s", display: "flex", alignItems: "center", gap: 8
            }}
          >
            <span style={{ fontSize: 18 }}>{l.icon}</span>
            <div>
              <div style={{ fontSize: 12, fontWeight: 700, color: leversActive[l.id] ? "#f0f0f0" : "#555" }}>{l.label}</div>
              <div style={{ fontSize: 11, color: leversActive[l.id] ? l.color : "#333" }}>+{l.impact}% impact</div>
            </div>
          </div>
        ))}
      </div>

      {/* Result */}
      <div style={{
        textAlign: "center", padding: "28px 20px",
        background: `linear-gradient(135deg, ${getColor(projected)}11, transparent)`,
        border: `2px solid ${getColor(projected)}`,
        borderRadius: 16, position: "relative", overflow: "hidden"
      }}>
        <div style={{ fontSize: 12, color: "#555", letterSpacing: 2, marginBottom: 4 }}>PROJECTED VIEWS PER VIDEO</div>
        <div style={{
          fontSize: 56, fontWeight: 900, color: getColor(projected),
          lineHeight: 1, marginBottom: 4,
          textShadow: `0 0 40px ${getColor(projected)}`
        }}>
          {projected >= 1000 ? `${(projected / 1000).toFixed(1)}K` : projected}
        </div>
        <div style={{ fontSize: 13, color: "#555" }}>
          {activeCount === 0 && "Turn on levers above to see your potential"}
          {activeCount === 1 && "Good start — stack more levers"}
          {activeCount === 2 && "Getting there — 2 more levers to go"}
          {activeCount === 3 && "Solid. One more and you're in viral territory"}
          {activeCount === 4 && "Almost there — add the last lever"}
          {activeCount === 5 && "🔥 All 5 levers active — maximum growth mode"}
        </div>
        {projected >= 5000 && (
          <div style={{ marginTop: 10, fontSize: 12, color: getColor(projected), opacity: 0.7 }}>

          </div>
        )}
      </div>
    </div>
  );
}
